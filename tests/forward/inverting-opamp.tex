\documentclass[border=12pt]{standalone}
\usepackage{amsmath}
\usepackage[american resistors,american inductors,american voltages,american currents]{circuitikz}
\begin{document}
\begin{circuitikz}
% Ideal signal-only op amp: supply pins intentionally omitted.
% Nets: IN = Rin.1; SUM = Rin.2 = Rf.1 = U1.-;
% OUT = U1.out = Rf.2; GND = U1.+.
\node[op amp] (U1) at (5,2) {$U_1$};
\coordinate (sum) at (2.5,2 |- U1.-);
\coordinate (input) at (0,2 |- U1.-);
\coordinate (out) at (7,2);
\draw (input) node[ocirc,label=left:$V_{\mathrm{in}}$] {}
      to[R,l={$R_{\mathrm{in}}=10\,\mathrm{k}\Omega$}] (sum)
      -- (U1.-);
\draw (U1.+) -- ++(-0.5,0) -- ++(0,-1) node[ground] {};
\draw (U1.out) -- (out) -- (8,2)
      node[ocirc,label=right:$V_{\mathrm{out}}$] {};
\draw (sum) node[circ] {} -- (sum |- 0,4)
      to[R,l={$R_f=100\,\mathrm{k}\Omega$}] (7,4)
      -- (out) node[circ] {};
\end{circuitikz}
